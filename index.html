/* * [1: HANDS MODEL] MediaPipe-MCU-Studio -> Arduino / ESP32 Serial Receiver * Packet Format: "H:1,D:120,C:5,F:11111\n" * H = Hand Detected Boolean (1 = True / 0 = False) * D = Thumb-Index Distance (px) * C = Open Fingers Count (0..5) * F = 5-Finger Binary State (Thumb, Index, Middle, Ring, Pinky) */ bool handDetected = false; int thumbIndexDist = 0; int openFingersCount = 0; String fingersState = "00000"; void setup() { Serial.begin(115200); Serial.println("MCU_READY: Waiting for MediaPipe HANDS Data..."); } String getValue(String data, String key) { int start = data.indexOf(key + ":"); if (start == -1) return ""; start += key.length() + 1; int end = data.indexOf(",", start); if (end == -1) end = data.length(); return data.substring(start, end); } void loop() { if (Serial.available() > 0) { String line = Serial.readStringUntil('\n'); line.trim(); String handStr = getValue(line, "H"); String distStr = getValue(line, "D"); String countStr = getValue(line, "C"); String statesStr = getValue(line, "F"); // e.g. "10100" } }
/* * [2: FACE MODEL] MediaPipe-MCU-Studio -> Arduino / ESP32 Serial Receiver * Packet Format: "FD:1,X:320,Y:240,W:180,T:-5\n" * FD = Face Detected Boolean (1 = True / 0 = False) * X = Face Center X Coordinate (px) * Y = Face Center Y Coordinate (px) * W = Face Box Width / Proximity (px) * T = Head Tilt Angle in Degrees (-45..+45) */ bool faceDetected = false; int faceX = 0; int faceY = 0; int faceWidth = 0; int headTilt = 0; void setup() { Serial.begin(115200); Serial.println("MCU_READY: Waiting for MediaPipe FACE Data..."); } String getValue(String data, String key) { int start = data.indexOf(key + ":"); if (start == -1) return ""; start += key.length() + 1; int end = data.indexOf(",", start); if (end == -1) end = data.length(); return data.substring(start, end); } void loop() { if (Serial.available() > 0) { String line = Serial.readStringUntil('\n'); line.trim(); String faceStr = getValue(line, "FD"); String xStr = getValue(line, "X"); String yStr = getValue(line, "Y"); String widthStr = getValue(line, "W"); String tiltStr = getValue(line, "T"); } }
/* * [3: FACE MESH MODEL] MediaPipe-MCU-Studio -> Arduino / ESP32 Serial Receiver * Packet Format: "MD:1,NX:320,NY:240,Y:10,P:-5,M:0,E:1\n" * MD = Mesh Detected Boolean (1 = True / 0 = False) * NX = Nose Tip X Coordinate (px) * NY = Nose Tip Y Coordinate (px) * Y = Head Yaw Left/Right Angle (-45..+45 deg) * P = Head Pitch Up/Down Angle (-45..+45 deg) * M = Mouth Open Boolean (1 = True [Open] / 0 = False [Closed]) * E = Both Eyes Open Boolean (1 = True [Open] / 0 = False [Closed]) */ bool meshDetected = false; int noseX = 0; int noseY = 0; int headYaw = 0; int headPitch = 0; bool mouthOpen = false; bool eyesOpen = false; void setup() { Serial.begin(115200); Serial.println("MCU_READY: Waiting for MediaPipe FACE MESH Data..."); } String getValue(String data, String key) { int start = data.indexOf(key + ":"); if (start == -1) return ""; start += key.length() + 1; int end = data.indexOf(",", start); if (end == -1) end = data.length(); return data.substring(start, end); } void loop() { if (Serial.available() > 0) { String line = Serial.readStringUntil('\n'); line.trim(); String meshStr = getValue(line, "MD"); String noseXStr = getValue(line, "NX"); String noseYStr = getValue(line, "NY"); String yawStr = getValue(line, "Y"); String pitchStr = getValue(line, "P"); String mouthStr = getValue(line, "M"); // "1" = True (Open), "0" = False (Closed) String eyesStr = getValue(line, "E"); // "1" = True (Open), "0" = False (Closed) } }
/* * [4: POSE MODEL] MediaPipe-MCU-Studio -> Arduino / ESP32 Serial Receiver * Packet Format: "PD:1,EL:160,ER:90,SL:45,SR:110,WD:250,AR:01\n" * PD = Body Detected Boolean (1 = True / 0 = False) * EL = Left Elbow Angle (0..180 deg) * ER = Right Elbow Angle (0..180 deg) * SL = Left Shoulder Elevation Angle (0..180 deg) * SR = Right Shoulder Elevation Angle (0..180 deg) * WD = Wrists Distance (px) * AR = Hands Raised State [LeftArm, RightArm] ("11"=Both Up, "00"=Both Down) */ bool bodyDetected = false; int leftElbow = 0; int rightElbow = 0; int leftShoulder = 0; int rightShoulder = 0; int wristsDist = 0; String armsState = "00"; void setup() { Serial.begin(115200); Serial.println("MCU_READY: Waiting for MediaPipe POSE Data..."); } String getValue(String data, String key) { int start = data.indexOf(key + ":"); if (start == -1) return ""; start += key.length() + 1; int end = data.indexOf(",", start); if (end == -1) end = data.length(); return data.substring(start, end); } void loop() { if (Serial.available() > 0) { String line = Serial.readStringUntil('\n'); line.trim(); String bodyStr = getValue(line, "PD"); String elLStr = getValue(line, "EL"); String elRStr = getValue(line, "ER"); String shLStr = getValue(line, "SL"); String shRStr = getValue(line, "SR"); String wDistStr = getValue(line, "WD"); String armsStr = getValue(line, "AR"); // e.g. "00", "10", "01", "11" } }
HOW TO USE SerialBridge.js IN YOUR OWN PROJECTS: Include ./js/serial-bridge.js in any HTML file to connect JavaScript with Arduino/ESP32.
// 1. Import the standalone library in HTML: // <script src="./js/serial-bridge.js"></script> // 2. Create a SerialBridge instance: const mcu = new SerialBridge({ baudRate: 115200, sendIntervalMs: 50, // Automatically throttles data to 20Hz onStatusChange: (connected, baud) => { console.log("Connected:", connected, "Baud:", baud); }, onReceive: (line) => { console.log("From Arduino:", line); } }); // 3. Connect from a button click: document.getElementById("myBtn").onclick = () => mcu.connect(115200); // 4. Send MediaPipe or custom commands anywhere in your loop: mcu.sendRaw("H:1,D:125,C:5,F:11111");
